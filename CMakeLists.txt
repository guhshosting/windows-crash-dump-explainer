cmake_minimum_required(VERSION 3.20)
project(windows-crash-dump-explainer VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Windows MSVC-compatible toolchain required")
endif()
option(WCDE_ANALYZE "MSVC analysis" OFF)
option(WCDE_ASAN "AddressSanitizer" OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(dump_parser src/parser.cpp)
target_include_directories(dump_parser PUBLIC src)
add_executable(windows-crash-dump-explainer src/main.cpp)
target_link_libraries(windows-crash-dump-explainer PRIVATE dump_parser)
add_executable(parser_tests tests/parser_tests.cpp)
target_link_libraries(parser_tests PRIVATE dump_parser)
foreach(t dump_parser windows-crash-dump-explainer parser_tests)
 target_compile_options(${t} PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf /sdl)
 target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
 if(WCDE_ANALYZE)
  target_compile_options(${t} PRIVATE /analyze)
 endif()
 if(WCDE_ASAN)
  target_compile_options(${t} PRIVATE /fsanitize=address /Zi)
  target_link_options(${t} PRIVATE /INCREMENTAL:NO)
 endif()
endforeach()
target_link_options(windows-crash-dump-explainer PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
enable_testing()
add_test(NAME parser COMMAND parser_tests)
add_test(NAME version COMMAND windows-crash-dump-explainer --version)
add_test(NAME help COMMAND windows-crash-dump-explainer --help)
install(TARGETS windows-crash-dump-explainer RUNTIME DESTINATION .)
install(FILES README.md LICENSE DESTINATION .)
install(DIRECTORY docs DESTINATION .)

find_program(WCDE_PWSH pwsh)
if(WCDE_PWSH)
 add_test(NAME cli_schema COMMAND "${WCDE_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli-tests.ps1" -Exe "$<TARGET_FILE:windows-crash-dump-explainer>" -FixtureTool "$<TARGET_FILE:parser_tests>" -Work "${CMAKE_CURRENT_BINARY_DIR}/cli-fixtures" -Schema "${CMAKE_CURRENT_SOURCE_DIR}/docs/schema.json")
 set_tests_properties(cli_schema PROPERTIES TIMEOUT 180)
endif()
set_tests_properties(parser PROPERTIES TIMEOUT 120)
option(WCDE_SDK_FIXTURE "Build optional test-only self-snapshot writer (x64)" OFF)
if(WCDE_SDK_FIXTURE)
 add_executable(sdk_fixture tests/sdk_fixture.cpp)
 target_link_libraries(sdk_fixture PRIVATE dbghelp)
 target_compile_options(sdk_fixture PRIVATE /W4 /WX /permissive- /utf-8 /sdl)
 target_compile_definitions(sdk_fixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
endif()
